测试程序 为了方便  要用的时候再定义

100万个随机数变成字符串放进去

比较两个方法 find 看运气 循序 和 bsearch(sort之后)
```cpp
#include <vector>
#include <stdexcept>
#include <string>
#include <cstdlib> //abort()
#include <iostream> //snprintf()
#include <ctime>
#include <algorithm> //sort()
namespace test02{

void test_vector(long& value){

	vector<string> c;// 没有元素，0
	
	char buf[10];
	clock_t timeStart= clock();
	for(long i=0;i<value;++i){
		
		//100万 万一内存不够
		try{
			snprintf(buf,10,"%d",rand());
			c.push_back(string(buf));
		}
		catch(exception&p){
			cout<<"i="<<i<<" "<<p.what()<<endl;
			abort();
		}
	}
	
	//可以呼叫的函数
	
	//capacity>size
	vector.size()
	vector.front()
	vector.back()
	vector.data()
	vector.capacity()
	
	
	string target=get_a_target_string();
	{
	
		timeStart=clock();
		
		//算法 模板函数find, 标准库 循序查找 双冒号 表示全局函数
		
		//auto   iterator 类型写起来太长
		auto pItem=::find(c.begin(),c.end(),target);
		cout<<"::find(),milli-seconds: "<<(clock()-timeStart)<<endl;
		if(pItem!=c.end())
			cout<<"found: "<<*pItem<<endl;
		else
			cout<<"not found!"<<endl;
	}
	
	
	timeStart=clock(); //程序从开始到现在
	sort(c.begin(),c.end());
	string* pItem=(string*)bsearch(&target,(c.data()),ASIZE,sizeof(string),comparestrings);
	cout<<"sort()+bsearch(),milli-seconds:"<<(clock()-timeStart)<<endl;
	
	if(pItem!=NULL)
		cout<<"found,"<<*pItem<<endl;
	else
		cout<<"not found!"<<endl;
	
	}

}
```
