## 原理

小和问题。

**特征**：
1. 思考一个问题在大范围上的答案，是否等于 左部分的答案+右部分的答案+跨越左右产生的答案。![[Pasted image 20260421150153.png]]


2. 在计算跨越左右产生的答案时，如果加上左、右各自有序这个设定，会不会获得计算的便利性。![[Pasted image 20260421150627.png]]
   指针只往右移动，不回退，从左划到右![[Pasted image 20260421150959.png]]
3.  若以上两点都成立，那么该问题可能被归并分治解决
4. 具体实施：求解过程中只需加入归并排序的过程即可，因为要让左右各自有序，来获得计算的便利性。

函数跑完最后既返回内部小和累加和产生多少，也保证小范围上有序。![[Pasted image 20260421151418.png]]
方便算跨左右的答案。


```java

//返回arr[l...r]范围上，小和的累加和，同时请把arr[l...r]变有序
public static long smallSum(int l,int r){
	if(l==r){
		return 0;//只有一个数字 没有小和
	}
	int m=(l+r)/2;
	return smallSum(l,m)+smallSum(m+1,r)+merge(l,m,r);

}

//返回跨左右产生的小和累加和，左侧有序，右侧有序，让左右两侧整体有序  
public static long merge(int l,int m,int r){
	//统计部分
	long ans=0;
	
	// 4 6 7 7    5 6 8 8
	// i          j
	//i的左侧小于等于j当前数字,求j来到的数字的答案
	//左侧划一遍 右侧划一遍
	for(int j=m+1,i=l,sum=0;j<=r;j++){
		while(i<=m&&arr[i]<=arr[j]){
			sum+=arr[i++];
		}
		ans+=sum;
	}
	
	//正常merge

	int a=l;

		help[i++]=arr[a]<=arr[b]?arr[a++]:arr[b++];
	}
	while(a<=m){
		help[i++]=arr[a++];
	}
	while(b<=r){
		help[i++]=arr[b++];
	}
	for(i=1;i<=r;i++){
		arr[i]=help[i];
	}

}
```

```java
// 统计l r翻转对数量，同时l r统计完后变有序

public static int counts(int l,int r){
	if(l==r){
		return 0;
	}
	int m=(l+r)/2;
	return counts(l,m)+counts(m+1,r)+merge(arr,l,m,r);
}

publci static int merge(int[]arr,int l,int m,int r){
	//统计部分
	int ans=0;
	
	//根据i 
	for(int i=l,j=m+1;i<=m;i++){
		//long是因为 *2的部分有可能溢出
		while(j<=r&&(long)arr[i]>(long)arr[j]*2){
			j++;
		}
		//j的左边有多少个数
		ans+=j-m-1;
	}

	//正常merge
	int i=l;
	int a=l;
	int b=m+1;
	while(a<=m&&b<=r){
		help[i++]=arr[a]<=arr[b]?arr[a++]:arr[b++];
	}
	while(a<=m){
		help[i++]=arr[a++];
	}
	while(b<=r){
		help[i++]=arr[b++];
	}
	for(i=l;i<=r;i++){
		arr[i]=help[i];
	}
	return ans;
}
```
